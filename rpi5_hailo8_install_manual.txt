1. Enable PCIe
$ sudo raspi-config
select [Advanced Options] => [PCIe Speed]=>[Yes]
Finish and reboot

2. update system
Add following line to /etc/apt/sources.list
deb http://archive.raspberrypi.org/debian/ buster main

$ sudo apt update && sudo apt full-upgrade
$ sudo rpi-eeprom-update

$ sudo apt install -y raspberrypi-kernel-headers dkms build-essential \
cmake git python3-dev python3-pip pybind11-dev \
rsync ffmpeg x11-utils \
python3-setuptools python3-virtualenv python-gi-dev libgirepository1.0-dev \
gcc-12 g++-12 libzmq3-dev pkg-config \
libcairo2-dev libgstreamer1.0-dev libgstreamer-plugins-base1.0-dev \
libgstreamer-plugins-bad1.0-dev gstreamer1.0-tools \
python3-full gstreamer1.0-plugins-base gstreamer1.0-plugins-good gstreamer1.0-plugins-bad \
gstreamer1.0-plugins-ugly gstreamer1.0-libav  gstreamer1.0-x \
gstreamer1.0-alsa gstreamer1.0-gl gstreamer1.0-gtk3 gstreamer1.0-qt5 \
gstreamer1.0-pulseaudio python3-gi python3-gi-cairo gir1.2-gtk-3.0

3. Enable Camrea
$ sudo apt install -y rpicam-apps
// camera config and test operations are written in other document

4. install Hailo-8 PCIe driver
$ git clone -b hailo8 https://github.com/hailo-ai/hailort-drivers.git
$ cd hailort-drivers/linux/pcie

$ make all
$ sudo make install
$ sudo modprobe hailo_pci

5. download Hailo-8 firmware
$ cd ../..
$ ./download_firmware.sh
$ sudo mkdir -p /lib/firmware/hailo
$ sudo mv hailo8_fw.<VERSION>.bin /lib/firmware/hailo/hailo8_fw.bin

$ sudo cp ./linux/pcie/51-hailo-udev.rules /etc/udev/rules.d/
$ sudo udevadm control --reload-rules && sudo udevadm trigger

6. install HailoRT
$ git clone -b hailo8 https://github.com/hailo-ai/hailort.git
$ cd hailort

$ cmake -H. -Bbuild -DCMAKE_BUILD_TYPE=Release
$ cmake --build build --config release -j$(nproc)
$ sudo cmake --build build --target install

$ mkdir build && cd build
$ cmake .. -DCMAKE_INSTALL_PREFIX=/usr -DHAILO_BUILD_GSTREAMER=ON -DCMAKE_INSTALL_LIBDIR=lib
$ make -j$(nproc)
$ sudo make install

$ sudo ldconfig

// should be able to find relative files
ls /usr/include/hailo/
ls /usr/lib/libhailort.so*

// reboot
check if NPU works well
$ hailortcli fw-control identify

7. install tappas
$ export TAPPAS_WORKSPACE=/home/hailo/Hailo/tappas

$ git clone https://github.com/hailo-ai/tappas.git
$ cd tappas
$ mkdir hailort
$ git clone https://github.com/hailo-ai/hailort.git -b hailo8 hailort/sources

$ ./install.sh --skip-hailort --target-platform rpi5
